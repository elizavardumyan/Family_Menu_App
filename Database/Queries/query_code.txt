---------------------------------
select unit table with translation units

---------------------------------
---------------------------------

SELECT
    u.unit_code,
    ut.language_code,
    ut.unit_name
FROM units AS u
JOIN unit_translations AS ut
    ON u.unit_id = ut.unit_id
ORDER BY
    u.unit_code,
    ut.language_code;

---------------------------------
---------------------------------
select group table with translation groups

---------------------------------
---------------------------------

SELECT
    pg.group_code,
	pg.group_id,
    pgt.language_code,
    pgt.group_name
FROM product_groups AS pg
JOIN product_group_translations AS pgt
    ON pg.group_id = pgt.group_id
ORDER BY
    pg.group_code,
    pgt.language_code;
